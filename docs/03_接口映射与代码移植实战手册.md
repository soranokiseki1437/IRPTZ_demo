### 2.1 会话与基础设备接口

| 功能 | SDK 函数签名（`SVNETSDK.h`） | 关键参数与官方 Demo 实证说明 |
| :--- | :--- | :--- |
| **全局初始化** | `LONG SV_NET_DEV_SDK_Init()` | 进程启动调一次 |

## 3. 核心模块代码实现原型

### 3.1 SDK 安全包装头：`svsdk.h`

`SVNETSDK.h` 在 Windows 下将 `SV_NET_SDK_API` 定义为 `__declspec(dllexport)` 而漏掉了 `extern "C"`，直接被 C++ 包含会导致链接时报 `undefined reference`。所有需要使用 SDK 的源文件统一 `#include "svsdk.h"`：

```cpp
#ifndef SVSDK_H
#define SVSDK_H

#ifdef _WIN32
#  ifndef WIN32_LEAN_AND_MEAN
#    define WIN32_LEAN_AND_MEAN
#  endif
#  ifndef NOMINMAX
#    define NOMINMAX
#  endif
#  include <winsock2.h>
#  include <windows.h>
#endif

extern "C" {
#include "SVNETSDK.h"
}

// 防止 Windows 宏污染 Qt / C++ 标准库
#ifdef min
#  undef min
#endif
#ifdef max
#  undef max
#endif

#endif // SVSDK_H
```

```cpp
// svsdksession.cpp
#include "svsdksession.h"
#include <QDebug>
#include <cstring>
#include <cstdlib>

SvSdkSession::SvSdkSession(QObject *parent) : QObject(parent) {}

SvSdkSession::~SvSdkSession()
{
    logout();
    if (m_inited) {
        SV_NET_DEV_SDK_Release();
        m_inited = false;
    }
}

bool SvSdkSession::ensureInit()
{
    QMutexLocker lock(&m_mutex);
    if (!m_inited) {
        SV_NET_DEV_SDK_Init();
        SV_NET_DEV_SetExceptionCallBack(0, nullptr, &SvSdkSession::onSdkException, this);
        m_inited = true;
    }
    return true;
}

bool SvSdkSession::parseAdvVer164(const char *softVer)
{
    if (!softVer) return true;
    const char *p = std::strstr(softVer, "V");
    if (!p) p = std::strstr(softVer, "v");
    if (!p) return true;
    int major = std::atoi(p + 1);
    const char *dot1 = std::strstr(p, ".");
    int minor = dot1 ? std::atoi(dot1 + 1) : 0;
    const char *dot2 = dot1 ? std::strstr(dot1 + 1, ".") : nullptr;
    int patch = dot2 ? std::atoi(dot2 + 1) : 0;
    return (major * 100 + minor * 10 + patch) >= 164;
}

bool SvSdkSession::login(const QString &ip, uint16_t port, const QString &user, const QString &pwd, QString *errOut)
{
    ensureInit();
    logout();

    SV_NET_DEV_USER_LOGIN_INFO loginInfo{};
    SV_NET_DEV_DEVICEINFO devInfo{};
    std::strncpy(loginInfo.sDeviceAddress, ip.toStdString().c_str(), SV_NET_DEV_ADDRESS_MAX_LEN - 1);
    loginInfo.wPort = (port == 0) ? 8000 : port;
    std::strncpy(loginInfo.sUserName, user.toStdString().c_str(), SV_NET_DEV_LOGIN_USERNAME_MAX_LEN - 1);
    std::strncpy(loginInfo.sPassword, pwd.toStdString().c_str(), SV_NET_DEV_LOGIN_PASSWD_MAX_LEN - 1);
    loginInfo.bUseAsynLogin = 0;

    LONG uid = SV_NET_DEV_Login(&loginInfo, &devInfo);
    if (uid < 0) {
        LONG err = SV_NET_DEV_GetLastError();
        if (errOut) *errOut = QString("SV_NET_DEV_Login 失败, 错误码=%1").arg(err);
        return false;
    }

    // 查询设备标准信息与固件版本（参照 sdkdemo.cpp:9418）
    SV_NET_DEV_STDINFO stdInfo{};
    if (SV_NET_DEV_GETSTDINFO(uid, &stdInfo) >= 0) {
        m_fwVersion = QString::fromLatin1(reinterpret_cast<const char*>(stdInfo.softwareVersion));
        bool adv = parseAdvVer164(reinterpret_cast<const char*>(stdInfo.softwareVersion));
        m_advVer164.store(adv);
        if (!adv) {
            // 旧版固件 PTZ 需先开启 RS485(2) 透明通道
            SV_NET_DEV_START_SERIALTRANS(uid, 2);
        }
    } else {
        m_advVer164.store(true);
    }

    m_userId.store(uid);
    qInfo() << "[SvSdkSession] 登录成功 userID=" << uid
            << " 固件版本=" << m_fwVersion
            << " AdvVer164=" << m_advVer164.load();
    return true;
}

void SvSdkSession::logout()
{
    LONG uid = m_userId.exchange(-1);
    if (uid >= 0) {
        SV_NET_DEV_Logout(uid);
    }
}

void CALLBACK SvSdkSession::onSdkException(DWORD dwType, LONG /*lUserID*/, LONG /*lHandle*/, void *pUser)
{
    auto *self = static_cast<SvSdkSession*>(pUser);
    if (!self) return;
    if (dwType == SUCCESS_RECONNECT) {
        emit self->reconnected();
    } else {
        emit self->networkException(dwType);
    }
}
```

```cpp
}

void NetPtzController::init()
{
    qInfo() << "[NetPtzController] 工作线程初始化完成, channel=" << m_channel;
}

void NetPtzController::setSpeedRange(int minSdkSpeed, int maxSdkSpeed)
{
    m_minSdkSpeed = qMax(0, minSdkSpeed);
    m_maxSdkSpeed = qMax(m_minSdkSpeed, maxSdkSpeed);
}

DWORD NetPtzController::mapPelcoSpeed(uint8_t pelcoSpeed) const
{
    if (pelcoSpeed == 0 || m_maxSdkSpeed == 0) return 0;
    float ratio = qBound(0.0f, pelcoSpeed / 63.0f, 1.0f);
    return static_cast<DWORD>(qRound(m_minSdkSpeed + ratio * (m_maxSdkSpeed - m_minSdkSpeed)));
}

void NetPtzController::sendSdkPtz(DWORD dwCmd, DWORD dwStop, DWORD dwSpeed)
{
    if (!isOpen()) return;
    LONG uid = m_session->userId();
    bool adv = m_session->isAdvVer164();

    if (dwStop == 1) {
        if (adv) {
            // 先按上一次运动方向发 stop=1，再发 cmd=0, stop=1 双重保险
            if (m_lastCmd != 0) {
                SV_NET_DEV_PTZControl_EX(uid, m_channel, m_lastCmd, 1);
            }
            SV_NET_DEV_PTZControl_EX(uid, m_channel, 0, 1);
        } else {
            DWORD stopCmd = (m_lastCmd != 0) ? m_lastCmd : UP_RIGHT;
            SV_NET_DEV_PTZControl(uid, stopCmd, 1);
        }
        return;
    }

    if (adv) {
        if (dwSpeed == 0) {
            SV_NET_DEV_PTZControl_EX(uid, m_channel, dwCmd, 0);
        } else {
            SV_NET_DEV_PTZControlWithSpeed_EX(uid, m_channel, dwCmd, 0, dwSpeed);
        }
    } else {
        if (dwSpeed == 0) {
            SV_NET_DEV_PTZControl(uid, dwCmd, 0);
        } else {
            SV_NET_DEV_PTZControlWithSpeed(uid, dwCmd, 0, dwSpeed);
        }
    }
}

void NetPtzController::moveDirection(uint8_t cmd2, uint8_t hSpeed, uint8_t vSpeed)
{
    if (!isOpen()) return;

    bool left  = (cmd2 & 0x04) != 0;
    bool right = (cmd2 & 0x02) != 0;
    bool up    = (cmd2 & 0x08) != 0;
    bool down  = (cmd2 & 0x10) != 0;

    // 主次轴抑制：SDK 对角线运动只接受单一速度值。
    // 若次轴速度不足主轴的 30%，优先只走主轴，防止次轴以主轴高速过冲。
    if ((left || right) && (up || down)) {
        if (vSpeed < hSpeed * 0.3f) {
            up = down = false;
        } else if (hSpeed < vSpeed * 0.3f) {
            left = right = false;
        }
    }

    DWORD dwCommand = 0;
    if (up && left)         dwCommand = UP_LEFT;
    else if (up && right)   dwCommand = UP_RIGHT;
    else if (down && left)  dwCommand = DOWN_LEFT;
    else if (down && right) dwCommand = DOWN_RIGHT;
    else if (up)            dwCommand = TILT_UP;
    else if (down)          dwCommand = TILT_DOWN;
    else if (left)          dwCommand = PAN_LEFT;
    else if (right)         dwCommand = PAN_RIGHT;

    if (dwCommand == 0) {
        stop();
        return;
    }

    // 每次收到有效运动指令，刷新 500ms 失联保护看门狗
    m_stopWatchdog->start(500);

    uint8_t rawSpeed = qMax(hSpeed, vSpeed);
    if (rawSpeed == 0) rawSpeed = 30;
    DWORD dwSpeed = mapPelcoSpeed(rawSpeed);

    // 指令去重：方向与速度均未改变时，不重复发起阻塞式网络请求
    if (m_isMoving && dwCommand == m_lastCmd && dwSpeed == m_lastSpeed) {
        return;
    }

    sendSdkPtz(dwCommand, 0, dwSpeed);
    m_lastCmd = dwCommand;
    m_lastSpeed = dwSpeed;
    m_isMoving = true;
}

void NetPtzController::stop()
{
    m_stopWatchdog->stop();
    if (!isOpen()) {
        m_isMoving = false;
        m_lastCmd = 0;
        m_lastSpeed = 0;
        return;
    }
    sendSdkPtz(m_lastCmd, 1, 0);
    m_isMoving = false;
    m_lastCmd = 0;
    m_lastSpeed = 0;
}

void NetPtzController::onStopWatchdogTimeout()

void NetPtzController::moveTo(float pan, float tilt, float /*speed*/)
{
    if (!isOpen() || !m_angleSupported) return;
    SV_NET_DEV_PTZ_POSITION pos{};
    pos.byEnable = 1;
    pos.struPtzPos.wAction = 1;
    float scale = (m_decodeMode == AngleDecodeMode::CentiDegree) ? 100.0f : 10.0f;
    float normPan = std::fmod(pan, 360.0f);
    if (normPan < 0) normPan += 360.0f;
    float normTilt = (tilt < 0) ? (360.0f + tilt) : tilt;
    pos.struPtzPos.wPanPos  = static_cast<WORD>(qRound(normPan * scale));
    pos.struPtzPos.wTiltPos = static_cast<WORD>(qRound(normTilt * scale));
    SV_NET_DEV_SetPtzPosition(m_session->userId(), m_channel, 0, &pos);
}

float NetPtzController::decodeRawAngle(WORD raw, bool isTilt) const
{
    float deg = 0.0f;
    if (m_decodeMode == AngleDecodeMode::HikBcd) {
        int d3 = (raw >> 12) & 0xF;
        int d2 = (raw >> 8)  & 0xF;
        int d1 = (raw >> 4)  & 0xF;
        int d0 = raw & 0xF;
        deg = (d3 * 1000 + d2 * 100 + d1 * 10 + d0) * 0.1f;
    } else if (m_decodeMode == AngleDecodeMode::CentiDegree) {
        deg = raw * 0.01f;
    } else {
        deg = raw * 0.1f;
    }
    if (isTilt && deg > 180.0f) {
        deg -= 360.0f; // 将无符号 WORD 的 270~360° 映射回 -90~0°
    }
    return deg;
}

void NetPtzController::queryAngle()
{
    if (!isOpen() || !m_angleSupported) return;

    SV_NET_DEV_PTZ_POSITION pos{};
    LONG ret = SV_NET_DEV_GetPtzPosition(m_session->userId(), m_channel, 0, &pos);
    if (ret < 0) {
        if (++m_queryFailCount >= 10) {
            m_angleSupported = false;
            m_autoQueryTimer->stop();
            m_autoQuerying.store(false);
            qWarning() << "[NetPtzController] GetPtzPosition 连续失败，已自动降级关闭角度轮询";
        }
        return;
    }
    m_queryFailCount = 0;

    float newPan  = decodeRawAngle(pos.struPtzPos.wPanPos, false);
    float newTilt = decodeRawAngle(pos.struPtzPos.wTiltPos, true);

    if (m_hasLastAngle) {
        float dt = m_angleTimer.restart() / 1000.0f;
        if (dt > 0.005f) {
            float dPan = newPan - m_currentPan;
            if (dPan > 180.0f) dPan -= 360.0f;
            if (dPan < -180.0f) dPan += 360.0f;
            float instPanSpeed  = dPan / dt;
            float instTiltSpeed = (newTilt - m_currentTilt) / dt;
            m_currentPanSpeed  = 0.7f * m_currentPanSpeed  + 0.3f * instPanSpeed;
            m_currentTiltSpeed = 0.7f * m_currentTiltSpeed + 0.3f * instTiltSpeed;
        }
    } else {
        m_angleTimer.start();
        m_hasLastAngle = true;
    }

    m_currentPan  = newPan;
    m_currentTilt = newTilt;
    emit angleReceived(m_currentPan, m_currentTilt, m_currentPanSpeed, m_currentTiltSpeed);
}

void NetPtzController::startAutoQuery(int intervalMs)

    m_autoQueryTimer->start(qMax(50, intervalMs));
    m_autoQuerying.store(true);

    m_autoQuerying.store(false);
}
```

```cpp
#include "DecodeThread.h"
#include "RecordThread.h"

#include <libavformat/avformat.h>
#include <libavcodec/avcodec.h>

    : QThread(parent), m_decoder(decoder) {}

{

    wait();
}

void RtspSource::setRecordThread(RecordThread *recorder, std::atomic<bool> *isRecording)
{
    m_recorder = recorder;
    m_isRecording = isRecording;
}

void RtspSource::stopStream()
{
    m_running.store(false);
}

void RtspSource::run()
{
    m_running.store(true);

    while (m_running.load()) {
        AVFormatContext *fmtCtx = avformat_alloc_context();
        AVDictionary *opts = nullptr;
        av_dict_set(&opts, "rtsp_transport", "tcp", 0);
        av_dict_set(&opts, "stimeout", "3000000", 0); // 3s 连接超时
        av_dict_set(&opts, "fflags", "nobuffer", 0);
        av_dict_set(&opts, "flags", "low_delay", 0);
        av_dict_set(&opts, "max_delay", "200000", 0);

        // 中断回调：使 stopStream() 能立即打断阻塞的 av_read_frame
        fmtCtx->interrupt_callback.callback = [](void *opaque) -> int {
            auto *running = static_cast<std::atomic<bool>*>(opaque);
            return running->load() ? 0 : 1;
        };
        fmtCtx->interrupt_callback.opaque = &m_running;

        if (avformat_open_input(&fmtCtx, m_url.toUtf8().constData(), nullptr, &opts) != 0) {
            av_dict_free(&opts);
            avformat_free_context(fmtCtx);
            emit streamError("RTSP 连接失败，2秒后重试");
            QThread::msleep(2000);
            continue;
        }
        av_dict_free(&opts);

        if (avformat_find_stream_info(fmtCtx, nullptr) < 0) {
            avformat_close_input(&fmtCtx);
            QThread::msleep(2000);
            continue;
        }

        int videoIdx = av_find_best_stream(fmtCtx, AVMEDIA_TYPE_VIDEO, -1, -1, nullptr, 0);
        if (videoIdx < 0) {
            avformat_close_input(&fmtCtx);
            emit streamError("RTSP 流中未找到视频轨");
            break;
        }

        AVCodecParameters *par = fmtCtx->streams[videoIdx]->codecpar;
        int encType = (par->codec_id == AV_CODEC_ID_HEVC) ? 3 : 0; // 与原工程约定一致：0=H264, 3=H265
        QByteArray extradata(reinterpret_cast<const char*>(par->extradata), par->extradata_size);

        if (m_decoder) {
            m_decoder->setEncType(encType);
        }
        emit streamConnected(encType);

        AVPacket *pkt = av_packet_alloc();
        while (m_running.load() && av_read_frame(fmtCtx, pkt) >= 0) {
            if (pkt->stream_index == videoIdx && pkt->size > 0) {
                QByteArray payload;
                if ((pkt->flags & AV_PKT_FLAG_KEY) && !extradata.isEmpty()) {
                    payload.reserve(extradata.size() + pkt->size);
                    payload.append(extradata);
                }
                payload.append(reinterpret_cast<const char*>(pkt->data), pkt->size);

                if (m_decoder) {
                    m_decoder->pushData(payload);
                }
                if (m_isRecording && m_isRecording->load() && m_recorder) {
                    m_recorder->pushData(payload);
                }
            }
            av_packet_unref(pkt);
        }

        av_packet_free(&pkt);
        avformat_close_input(&fmtCtx);
    }
}
```

### 4.2 六大实战避坑要点

1. **避坑 1：千万不要裸 `#include "SVNETSDK.h"`**
   - `SVNETSDK.h` 第 25 行未加 `extern "C"`，C++ 直接包含会按 C++ 规则修饰函数名，导致全部 SDK 函数链接失败。务必统一包含 `svsdk.h`。
2. **避坑 2：不要漏拷 `PlaySRenderCtrl.dll` 及其依赖的 `avcodec-58.dll` / `SDL2.dll` / `msvcr120.dll`**
   - 经二进制导入表核查，`SV_NET_SDK.dll` 静态依赖 `PlaySRenderCtrl.dll`、`pthreadVC2.dll`、`MSVCR120.dll`；而 `PlaySRenderCtrl.dll` 静态依赖 `avcodec-58.dll`、`SDL2.dll` 等。即使走 RTSP 不调用 `RealPlay`，只要加载 `SV_NET_SDK.dll`，Windows 加载器就会一并解析这些 DLL，缺任何一个都会报 `0xc0000135` / `0xc000007b`。
3. **避坑 3：严禁在 25Hz 跟踪循环中无条件调用 `SV_NET_DEV_PTZControlWithSpeed_EX`**
   - 该接口是同步网络请求。必须像 3.4 节 `NetPtzController::moveDirection` 那样，仅当 `(dwCommand, dwSpeed)` 发生变化时才调用 SDK，相同指令只重置 500ms 看门狗。
4. **避坑 4：`wTiltPos` 是无符号 `WORD`，不能直接除以 100 当作负数**
   - `SVNETSDK.h` 第 1098 行 `WORD wTiltPos` 是 `uint16_t`，负俯仰角通常以 `360° - |tilt|`（即 `> 180°`）表示，必须在解码后做 `if (deg > 180.0f) deg -= 360.0f;` 回绕处理。
5. **避坑 5：跟踪期间必须关闭自动 FFC 并锁定白热/黑热伪彩**
   - 自动打挡片会导致画面冻结约 0.5 秒，引发 PID 积分饱和与云台急冲；彩虹/铁红等非线性伪彩转灰度后会破坏目标梯度。已在 3.6 节 `InfraredService::lockTrackingPalette` 中内置联动保护。
6. **避坑 6：RTSP URL 中的密码特殊字符转义**
   - 若设备密码包含 `@`、`:`、`#` 等字符，拼装 `rtsp://user:pwd@ip:554/...` 时必须使用 `QUrl::toPercentEncoding(pwd)` 转义密码，否则 `avformat_open_input` 解析主机名会失败。
